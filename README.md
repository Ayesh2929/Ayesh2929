// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//   CLEAN ARCHITECTURE — DOMAIN LAYER
//   Zero external dependencies.
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// DOMAIN ENTITY — The heart of the system
class Message {
  private constructor(
    private readonly _id: MessageId,
    private readonly _conversationId: ConversationId,
    private readonly _senderId: UserId,
    private _content: MessageContent,
    private _status: MessageStatus,
    private readonly _createdAt: Date,
  ) {}

  static create(props: CreateMessageProps): Result<Message> {
    const contentResult = MessageContent.create(props.content);
    if (contentResult.isFailure)
      return Result.fail(contentResult.error);

    const message = new Message(
      MessageId.generate(),
      props.conversationId,
      props.senderId,
      contentResult.value,
      MessageStatus.SENT,
      new Date(),
    );

    message.addDomainEvent(new MessageSentEvent(message));
    return Result.ok(message);
  }

  markAsRead(): Result<void> {
    if (this._status.isSent())
      return Result.fail('Must be delivered first');
    this._status = MessageStatus.READ;
    this.addDomainEvent(new MessageReadEvent(this._id));
    return Result.ok();
  }
}

// VALUE OBJECT — Immutable, self-validating
class MessageContent {
  private constructor(private readonly _value: string) {}

  static create(content: string): Result<MessageContent> {
    if (!content?.trim())
      return Result.fail('Content cannot be empty');
    if (content.length > 10_000)
      return Result.fail('Exceeds 10,000 char limit');
    return Result.ok(new MessageContent(content.trim()));
  }
}

// PORT (Interface) — Dependency inversion
interface IMessageRepository {
  findById(id: MessageId): Promise<Option<Message>>;
  save(message: Message): Promise<void>;
}

// USE CASE — Single responsibility
class SendMessageUseCase {
  constructor(
    private readonly messageRepo: IMessageRepository,
    private readonly conversationRepo: IConversationRepository,
    private readonly eventBus: IDomainEventBus,
  ) {}

  async execute(dto: SendMessageDTO): Promise<Result<MessageDTO>> {
    const conversation = await this.conversationRepo
      .findById(ConversationId.create(dto.conversationId));

    if (conversation.isNone())
      return Result.fail('Conversation not found');

    if (!conversation.value.hasMember(UserId.create(dto.senderId)))
      return Result.fail('User is not a member');

    const messageResult = Message.create({
      conversationId: ConversationId.create(dto.conversationId),
      senderId: UserId.create(dto.senderId),
      content: dto.content,
    });

    if (messageResult.isFailure)
      return Result.fail(messageResult.error);

    await this.messageRepo.save(messageResult.value);
    await this.eventBus.publishAll(messageResult.value.domainEvents);

    return Result.ok(MessageMapper.toDTO(messageResult.value));
  }
}

// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
//   JAVA 21 — VIRTUAL THREADS + CQRS
//   Hexagonal Architecture Spring Boot 3
// ━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// COMMAND (Write side — CQRS)
public record SendMessageCommand(
    @NotBlank String conversationId,
    @NotBlank String senderId,
    @NotBlank @Size(max = 10_000) String content
) implements Command {}

// COMMAND HANDLER
@Component
@Transactional
@RequiredArgsConstructor
public class SendMessageCommandHandler
    implements CommandHandler<SendMessageCommand, MessageResponse> {

  private final MessageRepository messageRepository;
  private final ConversationRepository conversationRepository;
  private final ApplicationEventPublisher eventPublisher;

  @Override
  public MessageResponse handle(SendMessageCommand command) {
    var conversation = conversationRepository
        .findById(new ConversationId(command.conversationId()))
        .orElseThrow(() ->
            new ConversationNotFoundException(command.conversationId()));

    if (!conversation.hasMember(new UserId(command.senderId()))) {
      throw new UnauthorizedMemberException(
          "User is not a member of this conversation");
    }

    var message = Message.create(
        new ConversationId(command.conversationId()),
        new UserId(command.senderId()),
        MessageContent.of(command.content())
    );

    var saved = messageRepository.save(message);
    saved.getDomainEvents().forEach(eventPublisher::publishEvent);
    saved.clearDomainEvents();

    return messageMapper.toResponse(saved);
  }
}

// VIRTUAL THREADS CONFIG — Java 21
@Configuration
public class VirtualThreadConfig {

  @Bean
  public TomcatProtocolHandlerCustomizer<?> virtualThreads() {
    return handler -> handler.setExecutor(
        Executors.newVirtualThreadPerTaskExecutor());
  }
}

// STRUCTURED CONCURRENCY — Java 21
public MessageAggregateDTO fetchAggregate(String messageId) {
  try (var scope = new StructuredTaskScope.ShutdownOnFailure()) {
    var messageFuture     = scope.fork(() -> messageRepo.findById(messageId));
    var senderFuture      = scope.fork(() -> userRepo.findSender(messageId));
    var reactionsFuture   = scope.fork(() -> reactionRepo.findByMessage(messageId));
    var attachmentsFuture = scope.fork(() -> attachmentRepo.findByMessage(messageId));

    scope.join().throwIfFailed();

    return MessageAggregateDTO.builder()
        .message(messageFuture.get())
        .sender(senderFuture.get())
        .reactions(reactionsFuture.get())
        .attachments(attachmentsFuture.get())
        .build();
  }
}

// OUTBOX PATTERN — Guaranteed delivery
@Scheduled(fixedDelay = 1000)
@Transactional
public void processOutboxEvents() {
  try (var executor = Executors.newVirtualThreadPerTaskExecutor()) {
    var pending = outboxRepository.findPendingEvents(BATCH_SIZE);

    var futures = pending.stream()
        .map(event -> executor.submit(() -> publishEvent(event)))
        .toList();

    futures.forEach(f -> {
      try { f.get(5, SECONDS); }
      catch (Exception e) { handlePublishFailure(e); }
    });
  }
}
